def times11(n):
    """n × 11 by the Trachtenberg rule: each digit plus its neighbour, right to left."""
    digits = [0] + [int(c) for c in str(n)]
    out, carry = [], 0
    for k in range(len(digits) - 1, -1, -1):
        neighbour = digits[k + 1] if k + 1 < len(digits) else 0
        v = digits[k] + neighbour + carry
        out.append(v % 10)
